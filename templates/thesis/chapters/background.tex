\begin{definition}
  A \emph{metric} on a set $X$ is a function $d \colon X \times X \to [0, \infty)$ such that, for all
  $x, y, z \in X$, $d(x, y) = 0$ exactly when $x = y$, $d(x, y) = d(y, x)$ and
  \begin{equation}
    d(x, z) \le d(x, y) + d(y, z) .
    \label{eq:triangle}
  \end{equation}
\end{definition}
